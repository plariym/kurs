CREATE TABLE public.customers (
	customer_id serial4 NOT NULL,
	fio varchar(255) NOT NULL,
	passport varchar(255) NOT NULL,
	phone varchar(20) NOT NULL,
	address varchar(255) NOT NULL,
	CONSTRAINT customers_pk PRIMARY KEY (customer_id)
);


-- public.sales определение

-- Drop table

-- DROP TABLE public.sales;

CREATE TABLE public.sales (
	sale_id serial4 NOT NULL,
	kind_moto_id int4 NOT NULL,
	customer_id int4 NOT NULL,
	price int4 NOT NULL,
	"date" date NOT NULL,
	CONSTRAINT sales_pk PRIMARY KEY (sale_id)
);

	kind_moto_id serial4 NOT NULL,

	"class" varchar(255) NOT NULL,
	"Power(h/p)" int4 NOT NULL,
	price int4 NOT NULL,
	CONSTRAINT type_motorcycle_pk PRIMARY KEY (kind_moto_id)
